import { supabase } from '../config/supabase';

declare global {
  interface Window {
    loadPyodide: (options?: { indexURL?: string }) => Promise<any>;
  }
}

let pyodideInstance: any = null;
let initPromise: Promise<any> | null = null;

async function getPyodideInstance(): Promise<any> {
  if (pyodideInstance) return pyodideInstance;
  if (initPromise) return initPromise;

  initPromise = (async () => {
    if (typeof window.loadPyodide !== 'function') {
      throw new Error('Pyodide script failed to load from CDN.');
    }

    const pyodide = await window.loadPyodide({
      indexURL: 'https://cdn.jsdelivr.net/pyodide/v0.26.2/full/',
    });

    // Load pre-compiled binary packages required by pipeline.py
    await pyodide.loadPackage(['numpy', 'scikit-learn', 'joblib']);

    // Fetch the raw Python scripts from public assets
    const [pipelineCode, riskScorerCode] = await Promise.all([
      fetch('/py_agent/pipeline.py').then((res) => res.text()),
      fetch('/py_agent/risk_scorer.py').then((res) => res.text()),
    ]);

    // Create filesystem directories in the virtual WASM environment
    pyodide.FS.mkdirTree('/home/pyodide/ml/models');
    pyodide.FS.mkdirTree('/home/pyodide/engine');

    // Mount Isolation Forest joblib weights if available
    try {
      const modelBuffer = await fetch('/py_agent/models/isolation_forest.joblib').then((res) =>
        res.arrayBuffer()
      );
      pyodide.FS.writeFile(
        '/home/pyodide/ml/models/isolation_forest.joblib',
        new Uint8Array(modelBuffer)
      );
    } catch {
      // Model initializes synthetic baseline automatically if weights are absent
    }

    pyodide.FS.writeFile('/home/pyodide/ml/pipeline.py', pipelineCode);

    // Mock core.supabase client bridge with strict in-memory SQL filtering for .eq() and .neq()
    pyodide.runPython(`
import sys
sys.path.append('/home/pyodide')

class MockTable:
    def __init__(self, data):
        self._data = list(data)

    def select(self, *args, **kwargs):
        return self

    def eq(self, column, value):
        val_str = str(value).strip().lower()
        self._data = [
            row for row in self._data 
            if str(row.get(column, '')).strip().lower() == val_str
        ]
        return self

    def neq(self, column, value):
        val_str = str(value).strip().lower()
        self._data = [
            row for row in self._data 
            if str(row.get(column, '')).strip().lower() != val_str
        ]
        return self

    def order(self, column, desc=False):
        return self

    def limit(self, count):
        self._data = self._data[:count]
        return self

    def execute(self):
        class Res:
            def __init__(self, d):
                self.data = d
        return Res(self._data)

class MockSupabase:
    def __init__(self):
        self.payloads = {}

    def set_data(self, key, data):
        self.payloads[key] = data

    def table(self, name):
        return MockTable(self.payloads.get(name, []))

class MockCore:
    supabase = MockSupabase()

sys.modules['core'] = MockCore
    `);

    // Load unchanged risk_scorer logic into Python runtime
    pyodide.runPython(riskScorerCode);

    pyodideInstance = pyodide;
    return pyodide;
  })();

  return initPromise;
}

export async function runPythonRiskEvaluation(params: {
  intentId: number;
  sender: string;
  recipient: string;
  amount: number;
  assetAddress: string;
  dailyLimit?: number;
  purposeHashHex?: string;
  orgName?: string;
}) {
  const pyodide = await getPyodideInstance();
  const treasurerWallet = params.sender.trim();
  const recipientWallet = params.recipient.trim();

  // Scoped strictly to the active company / treasurer's disbursements
  const [orgMemberRes, txRes, globalRes] = await Promise.all([
    supabase
      .from('organization_members')
      .select('id, is_verified, wallet_address, org_name')
      .eq('wallet_address', recipientWallet)
      .limit(1),
    supabase
      .from('transactions_testnet')
      .select('id, intent_id, total_amount, created_at, status, note, description, from_wallet, to_wallet, cosigner_1_name, cosigner_2_name')
      .eq('from_wallet', treasurerWallet),
    supabase
      .from('transactions_testnet')
      .select('id, total_amount, created_at, status, to_wallet, from_wallet')
      .eq('from_wallet', treasurerWallet)
      .eq('status', 'executed')
      .order('created_at', { ascending: false })
      .limit(25),
  ]);

  // Combine and deduplicate safely without TypeScript errors
  const combinedList: Array<Record<string, any>> = [
    ...(txRes.data || []),
    ...(globalRes.data || [])
  ];

  const seenIds = new Set<string>();
  const uniqueTx: Array<Record<string, any>> = [];

  for (const item of combinedList) {
    const key = String(item.id || item.intent_id || Math.random());
    if (!seenIds.has(key)) {
      seenIds.add(key);
      uniqueTx.push(item);
    }
  }

  // Feed strictly scoped records to Pyodide
  pyodide.globals.set('_org_members_data', orgMemberRes.data || []);
  pyodide.globals.set('_tx_data', uniqueTx);

  pyodide.runPython(`
import sys
core = sys.modules['core']
core.supabase.set_data('organizations', _org_members_data.to_py() if hasattr(_org_members_data, 'to_py') else _org_members_data)
core.supabase.set_data('organization_members', _org_members_data.to_py() if hasattr(_org_members_data, 'to_py') else _org_members_data)
core.supabase.set_data('transactions_testnet', _tx_data.to_py() if hasattr(_tx_data, 'to_py') else _tx_data)
  `);

  // Execute calculate_composite_risk in Python
  const pyArgs = JSON.stringify({
    intent_id: params.intentId || 0,
    sender: treasurerWallet,
    recipient: recipientWallet,
    amount: params.amount,
    asset_address: params.assetAddress,
    daily_limit: params.dailyLimit || 50000.0,
    timestamp_epoch: Math.floor(Date.now() / 1000),
    purpose_hash_hex: params.purposeHashHex || '',
    org_name: params.orgName || 'none',
  });

  const evaluationResultJson = pyodide.runPython(`
import json
args = json.loads('''${pyArgs}''')

score, features, rationale, should_challenge = calculate_composite_risk(
    intent_id=args['intent_id'],
    sender=args['sender'],
    recipient=args['recipient'],
    amount=args['amount'],
    asset_address=args['asset_address'],
    daily_limit=args['daily_limit'],
    timestamp_epoch=args['timestamp_epoch'],
    purpose_hash_hex=args['purpose_hash_hex'],
    org_name=args['org_name']
)

json.dumps({
    "risk_score": round(score, 2),
    "feature_vector": [round(f, 4) for f in features],
    "rationale": rationale,
    "should_challenge": should_challenge
})
  `);

  return JSON.parse(evaluationResultJson);
}
